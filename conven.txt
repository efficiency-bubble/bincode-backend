Node type conventions (subject to change):
0 = uint32($0);
1 = uint64($0);
2 = node($0)+node($1);
3 = node($0)-node($1);
4 = unsigned node($0)*node($1);
5 = uint32(ARGV[$0]);
6 = uint64(ARGV[$0]);
7 = return node($0);
8 = callf(node($)...);

100 = uint32(SYMTAB[uint64($0)]);
101 = uint64(SYMTAB[uint64($0)]);
